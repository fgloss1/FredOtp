import { NextResponse } from "next/server";
import { supabase } from "@/lib/supabase";

const PAYSTACK_SECRET_KEY = process.env.PAYSTACK_SECRET_KEY || "";
const NGN_EXCHANGE_RATE = 1500; // 1 USD = 1,500 NGN

export async function POST(req: Request) {
  try {
    const body = await req.json().catch(() => ({}));
    const { userId, email, amountUSD } = body;

    const numAmount = Number(amountUSD) || 10;
    if (numAmount < 1) {
      return NextResponse.json(
        { error: "Minimum top-up amount is $1.00 USD" },
        { status: 400 }
      );
    }

    const amountNGN = numAmount * NGN_EXCHANGE_RATE;
    const amountInKobo = Math.round(amountNGN * 100);
    const reference = `nav_pay_${Date.now()}_${Math.floor(Math.random() * 1000)}`;

    const origin =
      req.headers.get("origin") ||
      req.headers.get("referer") ||
      process.env.NEXT_PUBLIC_APP_URL ||
      "http://localhost:3000";

    const cleanOrigin = origin.replace(/\/$/, "");

    // Check if valid UUID before database insert
    const isValidUuid =
      typeof userId === "string" &&
      /^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$/.test(
        userId
      );

    if (isValidUuid) {
      try {
        await supabase.from("transactions").insert({
          user_id: userId,
          amount_usd: numAmount,
          amount_local: amountNGN,
          currency: "NGN",
          payment_method: "Paystack (Bank Transfer / Card)",
          reference: reference,
          status: "pending",
        });
      } catch (dbErr) {
        console.warn("DB insert notice:", dbErr);
      }
    }

    // Demo Mode fallback if PAYSTACK_SECRET_KEY is not set in .env.local yet
    if (!PAYSTACK_SECRET_KEY) {
      const demoUrl = `${cleanOrigin}/api/paystack/verify?reference=${reference}&demo=true`;
      return NextResponse.json({
        success: true,
        authorization_url: demoUrl,
        reference: reference,
      });
    }

    // Call Official Paystack Initialization API
    const response = await fetch("https://api.paystack.co/transaction/initialize", {
      method: "POST",
      headers: {
        Authorization: `Bearer ${PAYSTACK_SECRET_KEY}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify({
        email: email || "customer@example.com",
        amount: amountInKobo,
        reference: reference,
        callback_url: `${cleanOrigin}/api/paystack/verify?reference=${reference}`,
        metadata: {
          user_id: isValidUuid ? userId : null,
          amount_usd: numAmount,
        },
      }),
    });

    const data = await response.json();

    if (!data.status) {
      return NextResponse.json(
        { error: data.message || "Paystack gateway error" },
        { status: 400 }
      );
    }

    return NextResponse.json({
      success: true,
      authorization_url: data.data.authorization_url,
      reference: reference,
    });
  } catch (err: any) {
    return NextResponse.json(
      { error: err.message || "Internal server error" },
      { status: 500 }
    );
  }
}
