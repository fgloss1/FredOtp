A. Drop Old RPC + Create New Atomic RPC
SQL

-- Drop old function
DROP FUNCTION IF EXISTS add_balance_atomic(NUMERIC);

-- Create new atomic function (credits balance + completes transaction)
CREATE OR REPLACE FUNCTION complete_deposit_atomic(
  transaction_id UUID,
  amount_to_add NUMERIC
)
RETURNS NUMERIC AS $$
DECLARE
  new_balance NUMERIC;
  authenticated_user_id UUID;
  txn_user_id UUID;
  txn_status TEXT;
BEGIN
  -- Get authenticated user from Supabase auth context
  authenticated_user_id := auth.uid();
  
  -- Security check: Must be authenticated
  IF authenticated_user_id IS NULL THEN
    RAISE EXCEPTION 'Authentication required';
  END IF;
  
  -- Security check: Amount must be positive
  IF amount_to_add IS NULL OR amount_to_add <= 0 THEN
    RAISE EXCEPTION 'Invalid amount';
  END IF;
  
  -- Set safe search_path for SECURITY DEFINER function
  SET LOCAL search_path = public;
  
  -- Verify transaction exists, belongs to user, and is pending
  SELECT user_id, status INTO txn_user_id, txn_status
  FROM transactions
  WHERE id = transaction_id;
  
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Transaction not found';
  END IF;
  
  IF txn_user_id IS DISTINCT FROM authenticated_user_id THEN
    RAISE EXCEPTION 'Transaction does not belong to authenticated user';
  END IF;
  
  IF txn_status != 'pending' THEN
    RAISE EXCEPTION 'Transaction is not in pending status (current: %)', txn_status;
  END IF;
  
  -- ATOMIC OPERATION: Credit balance AND mark transaction completed
  -- Both operations succeed or both fail (same transaction)
  
  -- Step 1: Credit user's balance
  UPDATE profiles
  SET balance = (balance + amount_to_add)::NUMERIC
  WHERE id = authenticated_user_id
  RETURNING balance INTO new_balance;
  
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Profile not found for authenticated user';
  END IF;
  
  -- Step 2: Mark transaction as completed
  UPDATE transactions
  SET status = 'completed'
  WHERE id = transaction_id AND user_id = authenticated_user_id AND status = 'pending';
  
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Failed to update transaction status';
  END IF;
  
  RETURN new_balance;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Grant execute permission ONLY to authenticated users
REVOKE ALL ON FUNCTION complete_deposit_atomic(UUID, NUMERIC) FROM PUBLIC;
REVOKE ALL ON FUNCTION complete_deposit_atomic(UUID, NUMERIC) FROM anon;
GRANT EXECUTE ON FUNCTION complete_deposit_atomic(UUID, NUMERIC) TO authenticated;
