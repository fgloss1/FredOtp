1. New Table: deposit_intents

CREATE TABLE deposit_intents (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  coin VARCHAR(50) NOT NULL,
  network VARCHAR(50) NOT NULL,
  expected_amount_usd NUMERIC(12,2) NOT NULL,
  destination_address VARCHAR(255) NOT NULL,
  status VARCHAR(50) DEFAULT 'pending' NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW() NOT NULL,
  expires_at TIMESTAMPTZ NOT NULL,
  completed_at TIMESTAMPTZ,
  completed_tx_reference VARCHAR(255) UNIQUE -- Links to transactions.reference
);

-- Indexes for performance
CREATE INDEX idx_deposit_intents_user_id ON deposit_intents(user_id);
CREATE INDEX idx_deposit_intents_status ON deposit_intents(status);
CREATE INDEX idx_deposit_intents_expires_at ON deposit_intents(expires_at);

3. New RPC Function: complete_deposit_atomic()
SQL

CREATE OR REPLACE FUNCTION complete_deposit_atomic(
  intent_id UUID,
  transaction_id UUID,
  amount_to_add NUMERIC
)
RETURNS NUMERIC AS $$
DECLARE
  new_balance NUMERIC;
  authenticated_user_id UUID;
  intent_user_id UUID;
  intent_status TEXT;
  intent_created_at TIMESTAMPTZ;
  txn_user_id UUID;
  txn_status TEXT;
  txn_block_timestamp TIMESTAMPTZ;
BEGIN
  -- Get authenticated user
  authenticated_user_id := auth.uid();
  
  IF authenticated_user_id IS NULL THEN
    RAISE EXCEPTION 'Authentication required';
  END IF;
  
  IF amount_to_add IS NULL OR amount_to_add <= 0 THEN
    RAISE EXCEPTION 'Invalid amount';
  END IF;
  
  SET LOCAL search_path = public;
  
  -- Lock and verify deposit intent (FOR UPDATE prevents concurrent modification)
  SELECT user_id, status, created_at 
  INTO intent_user_id, intent_status, intent_created_at
  FROM deposit_intents
  WHERE id = intent_id
  FOR UPDATE;
  
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Deposit intent not found';
  END IF;
  
  IF intent_user_id IS DISTINCT FROM authenticated_user_id THEN
    RAISE EXCEPTION 'Deposit intent does not belong to authenticated user';
  END IF;
  
  IF intent_status != 'pending' THEN
    RAISE EXCEPTION 'Deposit intent is not pending (current: %)', intent_status;
  END IF;
  
  -- Lock and verify transaction
  SELECT user_id, status, block_timestamp
  INTO txn_user_id, txn_status, txn_block_timestamp
  FROM transactions
  WHERE id = transaction_id
  FOR UPDATE;
  
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Transaction not found';
  END IF;
  
  IF txn_user_id IS DISTINCT FROM authenticated_user_id THEN
    RAISE EXCEPTION 'Transaction does not belong to authenticated user';
  END IF;
  
  IF txn_status != 'pending' THEN
    RAISE EXCEPTION 'Transaction is not pending (current: %)', txn_status;
  END IF;
  
  -- REPLAY PROTECTION: Blockchain tx must be after intent creation
  IF txn_block_timestamp IS NOT NULL AND txn_block_timestamp < intent_created_at THEN
    RAISE EXCEPTION 'Blockchain transaction occurred before deposit intent. Historical replay detected.';
  END IF;
  
  -- ATOMIC: Credit balance (NO updated_at)
  UPDATE profiles
  SET balance = (balance + amount_to_add)::NUMERIC
  WHERE id = authenticated_user_id
  RETURNING balance INTO new_balance;
  
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Profile not found';
  END IF;
  
  -- ATOMIC: Mark transaction completed
  UPDATE transactions
  SET status = 'completed'
  WHERE id = transaction_id AND user_id = authenticated_user_id;
  
  -- ATOMIC: Mark intent completed
  UPDATE deposit_intents
  SET status = 'completed', completed_at = NOW(), completed_tx_reference = (SELECT reference FROM transactions WHERE id = transaction_id)
  WHERE id = intent_id AND user_id = authenticated_user_id;
  
  RETURN new_balance;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Permissions
REVOKE ALL ON FUNCTION complete_deposit_atomic(UUID, UUID, NUMERIC) FROM PUBLIC;
REVOKE ALL ON FUNCTION complete_deposit_atomic(UUID, UUID, NUMERIC) FROM anon;
GRANT EXECUTE ON FUNCTION complete_deposit_atomic(UUID, UUID, NUMERIC) TO authenticated;

Endpoint 1: POST /api/crypto/deposit-intent

// Creates deposit intent BEFORE user sends crypto
export async function POST(req: Request) {
  const { user } = await supabase.auth.getUser();
  const { coin, amountUsd } = await req.json();
  
  // Validate amount, coin, etc.
  
  const intent = await supabase
    .from("deposit_intents")
    .insert({
      user_id: user.id,
      coin,
      network: getNetwork(coin),
      expected_amount_usd: amountUsd,
      destination_address: process.env[`CRYPTO_${coin}`],
      status: "pending",
      expires_at: new Date(Date.now() + 24 * 60 * 60 * 1000),
    })
    .select()
    .single();
  
  return NextResponse.json({
    intent_id: intent.id,
    destination_address: intent.destination_address,
    expires_at: intent.expires_at,
  });
}
