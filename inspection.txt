   4: export async function POST(req: NextRequest) {
   5:   try {
   6:     const authHeader = req.headers.get('authorization');
   7:     if (!authHeader || !authHeader.startsWith('Bearer ')) {
   8:       return NextResponse.json({ error: 'Unauthorized: Session missing' }, { status: 401 });
   9:     }
  10:
  11:     const token = authHeader.replace('Bearer ', '').trim();
  12:     const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL || '';
  13:     const supabaseAnonKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY || '';
  14:     const serviceRoleKey = process.env.SUPABASE_SERVICE_ROLE_KEY || supabaseAnonKey;
  15:
  16:     const supabaseAnon = createClient(supabaseUrl, supabaseAnonKey);
  17:     const { data: { user }, error: userError } = await supabaseAnon.auth.getUser(token);
  18:
  19:     if (userError || !user) {
  20:       return NextResponse.json({ error: 'Unauthorized: Invalid or expired session' }, { status: 401 });
  21:     }
  22:
  23:     const body = await req.json();
  24:     const { planId, zipCode } = body;
  25:
  26:     if (!planId) {
  27:       return NextResponse.json({ error: 'Plan ID required' }, { status: 400 });
  28:     }
  29:
  30:     const db = createClient(supabaseUrl, serviceRoleKey);
  31:
  32:     const { data: order, error: insertError } = await db
  33:       .from('nava_esim_orders')
  34:       .insert({
  35:         user_id: user.id,
  36:         plan_id: planId,
  37:         preferred_zip: zipCode || '33917',
  38:         status: 'pending_payment',
  39:         created_at: new Date().toISOString()
  40:       })
  41:       .select('id')

  43:
  44:     if (insertError || !order) {

  46:       return NextResponse.json({ error: insertError?.message || 'Failed to create order' }, { status: 500 });
  47:     }
  48:
  49:     return NextResponse.json({ orderId: order.id, success: true });
  50:   } catch (err: any) {
  51:     console.error('Order Exception:', err);
  52:     return NextResponse.json({ error: err.message || 'Internal server error' }, { status: 500 });
