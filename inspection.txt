When a customer selects a ZIP code at checkout, the quote generator calculates or confirms the exact location tax/fee before generating the pending_payment order. Revised Migration SQL (20261007000000_nava_esim_init.sql)
SQL

-- =====================================================================
-- NAVA eSIM — Stage A: Additive Schema Initialization
-- Timestamp: 20261007000000 (Runs AFTER 20261006_otp_wallet_security.sql)
--
-- DOES NOT modify existing OTP, 5SIM, rentals, wallet, or auth tables.
-- Uses Supabase RLS and server-side RPC patterns consistent with
-- the existing public.profiles.balance wallet.
-- =====================================================================

CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- ---------------------------------------------------------------------
-- 1. eSIM Plans Catalogue
-- Configurable pricing in integer cents.
-- supports_zip = true for Nexitel plans allowing state/ZIP selection.
-- is_purchasable = false by default (admin master switch per plan).
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.nava_esim_plans (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  supplier TEXT NOT NULL DEFAULT 'nexitel',
  carrier TEXT NOT NULL,                        -- e.g. 'purple'
  code TEXT NOT NULL UNIQUE,                    -- e.g. 'purple_text', 'purple_max'
  name TEXT NOT NULL,
  description TEXT NOT NULL DEFAULT '',
  talk_minutes INTEGER,                          -- NULL = unlimited
  text_allowance TEXT NOT NULL DEFAULT 'unlimited',
  data_mb INTEGER,                               -- NULL = none / per product rules
  supports_zip BOOLEAN NOT NULL DEFAULT true,    -- Nexitel Purple supports ZIP selection
  duration_days INTEGER NOT NULL DEFAULT 30 CHECK (duration_days > 0),
  
  -- Baseline Activation Quote (cents)
  activation_plan_cents INTEGER NOT NULL CHECK (activation_plan_cents >= 0),
  activation_regulatory_fee_cents INTEGER NOT NULL DEFAULT 0 CHECK (activation_regulatory_fee_cents >= 0),
  activation_tax_cents INTEGER NOT NULL DEFAULT 0 CHECK (activation_tax_cents >= 0),
  
  -- Baseline Renewal Quote (cents)
  renewal_plan_cents INTEGER NOT NULL CHECK (renewal_plan_cents >= 0),
  renewal_regulatory_fee_cents INTEGER NOT NULL DEFAULT 0 CHECK (renewal_regulatory_fee_cents >= 0),
  renewal_tax_cents INTEGER NOT NULL DEFAULT 0 CHECK (renewal_tax_cents >= 0),
  
  price_source TEXT NOT NULL,                    -- e.g. 'manual_verified_checkout_2026-01-15'
  price_verified_at TIMESTAMPTZ,
  fulfillment_method TEXT NOT NULL DEFAULT 'manual'
    CHECK (fulfillment_method IN ('manual', 'api')),
  is_visible BOOLEAN NOT NULL DEFAULT true,
  is_purchasable BOOLEAN NOT NULL DEFAULT false,  -- Restricted until explicitly enabled
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS nava_esim_plans_visible_idx
  ON public.nava_esim_plans (is_visible, is_purchasable);

-- ---------------------------------------------------------------------
-- 2. eSIM Orders
-- Handles both Activations and Renewals with frozen price snapshots.
-- Records selected ZIP code for location-based taxation.
-- Status transitions enforce: payment SUCCESS != active line.
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.nava_esim_orders (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES auth.users (id) ON DELETE RESTRICT,
  plan_id UUID NOT NULL REFERENCES public.nava_esim_plans (id) ON DELETE RESTRICT,
  order_type TEXT NOT NULL CHECK (order_type IN ('activation', 'renewal')),
  parent_order_id UUID REFERENCES public.nava_esim_orders (id) ON DELETE RESTRICT,
  zip_code TEXT,                                -- Customer selected ZIP code

  -- Frozen Price Snapshot at time of order (cents)
  plan_price_cents INTEGER NOT NULL CHECK (plan_price_cents >= 0),
  regulatory_fee_cents INTEGER NOT NULL CHECK (regulatory_fee_cents >= 0),
  tax_cents INTEGER NOT NULL CHECK (tax_cents >= 0),
  total_cents INTEGER NOT NULL CHECK (total_cents >= 0),
  currency TEXT NOT NULL DEFAULT 'USD',

  -- Payment & Idempotency
  idempotency_key TEXT NOT NULL,
  wallet_debit_ref TEXT,                         -- Ref returned by otp_balance_adjust_atomic

  -- Order Fulfillment Status State Machine
  status TEXT NOT NULL DEFAULT 'pending_payment'
    CHECK (status IN (
      'pending_payment',
      'paid_awaiting_fulfillment',
      'processing',
      'active',
      'renewal_pending',
      'renewal_processing',
      'renewal_completed',
      'failed_requires_review',
      'refunded'
    )),

  -- Fulfillment Details (Manual Admin Fulfillment v1)
  assigned_phone_number TEXT,                    -- Real assigned number (E.164)
  activation_starts_at TIMESTAMPTZ,
  activation_expires_at TIMESTAMPTZ,
  fulfilled_by UUID REFERENCES auth.users (id),
  fulfilled_at TIMESTAMPTZ,
  supplier_confirmation TEXT,                    -- Admin confirmation note / order ID
  failure_reason TEXT,

  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),

  CONSTRAINT nava_esim_orders_idempotency
    UNIQUE (user_id, idempotency_key),

  CONSTRAINT nava_esim_orders_total_matches
    CHECK (total_cents = plan_price_cents + regulatory_fee_cents + tax_cents),

  CONSTRAINT nava_esim_orders_renewal_parent
    CHECK (
      (order_type = 'activation' AND parent_order_id IS NULL)
      OR
      (order_type = 'renewal' AND parent_order_id IS NOT NULL)
    )
);

CREATE INDEX IF NOT EXISTS nava_esim_orders_user_idx
  ON public.nava_esim_orders (user_id, created_at DESC);

CREATE INDEX IF NOT EXISTS nava_esim_orders_status_idx
  ON public.nava_esim_orders (status);

CREATE INDEX IF NOT EXISTS nava_esim_orders_expiry_active_idx
  ON public.nava_esim_orders (activation_expires_at)
  WHERE status = 'active';

-- ---------------------------------------------------------------------
-- 3. eSIM Order Secrets (QR Code & Provisioning Credentials)
-- Denied to direct client SELECT via RLS.
-- Served exclusively through server routes that verify user ownership
-- and status = 'active', producing short-lived signed URLs for storage.
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.nava_esim_order_secrets (
  order_id UUID PRIMARY KEY
    REFERENCES public.nava_esim_orders (id) ON DELETE CASCADE,
  activation_code_encrypted TEXT,                -- App-level encrypted string
  qr_storage_bucket TEXT NOT NULL DEFAULT 'nava-esim-qr',
  qr_storage_path TEXT,                          -- Private object path inside bucket
  customer_instructions TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- ---------------------------------------------------------------------
-- 4. Renewal Reminders Log (Idempotent Tracking)
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.nava_esim_reminders (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  order_id UUID NOT NULL
    REFERENCES public.nava_esim_orders (id) ON DELETE CASCADE,
  reminder_type TEXT NOT NULL
    CHECK (reminder_type IN ('expiry_7d', 'expiry_3d', 'expiry_1d')),
  channel TEXT NOT NULL CHECK (channel IN ('email', 'inapp')),
  sent_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (order_id, reminder_type, channel)
);

-- ---------------------------------------------------------------------
-- 5. Updated At Trigger
-- ---------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.nava_esim_set_updated_at()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
  NEW.updated_at = now();
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS nava_esim_plans_updated_at ON public.nava_esim_plans;
CREATE TRIGGER nava_esim_plans_updated_at
  BEFORE UPDATE ON public.nava_esim_plans
  FOR EACH ROW EXECUTE FUNCTION public.nava_esim_set_updated_at();

DROP TRIGGER IF EXISTS nava_esim_orders_updated_at ON public.nava_esim_orders;
CREATE TRIGGER nava_esim_orders_updated_at
  BEFORE UPDATE ON public.nava_esim_orders
  FOR EACH ROW EXECUTE FUNCTION public.nava_esim_set_updated_at();

DROP TRIGGER IF EXISTS nava_esim_order_secrets_updated_at ON public.nava_esim_order_secrets;
CREATE TRIGGER nava_esim_order_secrets_updated_at
  BEFORE UPDATE ON public.nava_esim_order_secrets
  FOR EACH ROW EXECUTE FUNCTION public.nava_esim_set_updated_at();

  ON public.nava_esim_orders
  FOR SELECT
  TO authenticated
