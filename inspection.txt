CREATE TABLE IF NOT EXISTS deposit_intents (

  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  coin VARCHAR(50) NOT NULL,
  network VARCHAR(50) NOT NULL,
  expected_amount_usd NUMERIC(12,2) NOT NULL,
  destination_address VARCHAR(255) NOT NULL,
  status VARCHAR(50) DEFAULT 'pending' NOT NULL,
  expires_at TIMESTAMPTZ NOT NULL DEFAULT (NOW() + INTERVAL '24 hours'),
  created_at TIMESTAMPTZ DEFAULT NOW() NOT NULL,
  completed_at TIMESTAMPTZ,
  completed_tx_reference VARCHAR(255) UNIQUE -- Links to transactions.reference
);

-- 3. Index for performance

B. New Atomic RPC Function (With Replay Protection)
SQL

-- Drop old function if exists
DROP FUNCTION IF EXISTS add_balance_atomic(NUMERIC);
DROP FUNCTION IF EXISTS complete_deposit_atomic(UUID, NUMERIC);

-- Create new atomic function with replay protection
CREATE OR REPLACE FUNCTION complete_deposit_atomic(
  transaction_id UUID,
  amount_to_add NUMERIC
)
RETURNS NUMERIC AS $$
DECLARE
  new_balance NUMERIC;
  authenticated_user_id UUID;
  txn_user_id UUID;
  txn_status TEXT;
  txn_block_timestamp TIMESTAMPTZ;
  txn_reference TEXT;
  intent_created_at TIMESTAMPTZ;
BEGIN
  -- Get authenticated user from Supabase auth context
  authenticated_user_id := auth.uid();
  
  -- Security check: Must be authenticated
  IF authenticated_user_id IS NULL THEN
    RAISE EXCEPTION 'Authentication required';
  END IF;
  
  -- Security check: Amount must be positive
  IF amount_to_add IS NULL OR amount_to_add <= 0 THEN
    RAISE EXCEPTION 'Invalid amount';
  END IF;
  
  -- Set safe search_path for SECURITY DEFINER function
  SET LOCAL search_path = public;
  
  -- Verify transaction exists, belongs to user, and is pending
  -- Also fetch block_timestamp and reference for replay protection
  SELECT user_id, status, block_timestamp, reference 
  INTO txn_user_id, txn_status, txn_block_timestamp, txn_reference
  FROM transactions
  WHERE id = transaction_id
  FOR UPDATE; -- Row lock to prevent concurrent modification
  
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Transaction not found';
  END IF;
  
  IF txn_user_id IS DISTINCT FROM authenticated_user_id THEN
    RAISE EXCEPTION 'Transaction does not belong to authenticated user';
  END IF;
  
  IF txn_status != 'pending' THEN
    RAISE EXCEPTION 'Transaction is not in pending status (current: %)', txn_status;
  END IF;
  
  -- REPLAY PROTECTION: Verify blockchain transaction occurred AFTER deposit intent
  -- Look for matching deposit intent (same user, same TxHash reference)
  SELECT created_at INTO intent_created_at
  FROM deposit_intents
  WHERE user_id = authenticated_user_id 
    AND completed_tx_reference = txn_reference
    AND status = 'pending'
    AND expires_at > NOW();
  
  IF NOT FOUND THEN
    RAISE EXCEPTION 'No valid deposit intent found for this transaction. Historical payment replay detected.';
  END IF;
  
  -- REPLAY PROTECTION: Blockchain tx must be after intent creation
  IF txn_block_timestamp IS NOT NULL AND txn_block_timestamp < intent_created_at THEN
    RAISE EXCEPTION 'Blockchain transaction occurred before deposit intent was created. Historical payment replay detected.';
  END IF;
  
  -- ATOMIC OPERATION: Credit balance AND mark transaction completed
  -- Both operations succeed or both fail (same PostgreSQL transaction)
  
  -- Step 1: Credit user's balance (NO updated_at column)
  UPDATE profiles
  SET balance = (balance + amount_to_add)::NUMERIC
  WHERE id = authenticated_user_id
  RETURNING balance INTO new_balance;
  
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Profile not found for authenticated user';
  END IF;
  
  -- Step 2: Mark transaction as completed
  UPDATE transactions
  SET status = 'completed'
  WHERE id = transaction_id 
    AND user_id = authenticated_user_id 
    AND status = 'pending';
  
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Failed to update transaction status';
  END IF;
  
  -- Step 3: Mark deposit intent as completed
  UPDATE deposit_intents
  SET status = 'completed',
      completed_at = NOW()
  WHERE user_id = authenticated_user_id 
    AND completed_tx_reference = txn_reference
    AND status = 'pending';
  
  RETURN new_balance;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Grant execute permission ONLY to authenticated users
REVOKE ALL ON FUNCTION complete_deposit_atomic(UUID, NUMERIC) FROM PUBLIC;
REVOKE ALL ON FUNCTION complete_deposit_atomic(UUID, NUMERIC) FROM anon;
GRANT EXECUTE ON FUNCTION complete_deposit_atomic(UUID, NUMERIC) TO authenticated;
