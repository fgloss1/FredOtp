$layoutTsx = @'
'use client';

import React, { useState, useEffect } from 'react';
import Link from 'next/link';
import { usePathname, useRouter } from 'next/navigation';
import { createClient } from '@supabase/supabase-js';

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL || '',
  process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY || ''
);

interface NavItem {
  label: string;
  href: string;
  icon: string;
  badge?: string;
}

interface NavGroup {
  title: string;
  items: NavItem[];
}

export default function DashboardLayout({ children }: { children: React.ReactNode }) {
  const pathname = usePathname();
  const router = useRouter();
  const [balance, setBalance] = useState<string>('0.00');
  const [userEmail, setUserEmail] = useState<string>('');
  const [isAdmin, setIsAdmin] = useState<boolean>(false);
  const [mobileOpen, setMobileOpen] = useState<boolean>(false);

  useEffect(() => {
    async function loadUser() {
      const { data: { session } } = await supabase.auth.getSession();
      if (!session) return;

      setUserEmail(session.user.email || '');

      // Load balance from profiles table
      const { data: profile } = await supabase
        .from('profiles')
        .select('balance, role, username')
        .eq('id', session.user.id)
        .single();

      if (profile) {
        setBalance(Number(profile.balance || 0).toFixed(2));
        setIsAdmin(profile.role === 'admin' || profile.role === 'superadmin');
      }
    }

    loadUser();
  }, [pathname]);

  const navGroups: NavGroup[] = [
    {
      title: 'Overview',
      items: [
        { label: 'Dashboard', href: '/dashboard', icon: '🏠' }
      ]
    },
    {
      title: 'Numbers & Verification',
      items: [
        { label: 'Get OTP Number', href: '/dashboard/rentals', icon: '⚡' },
        { label: 'OTP Order History', href: '/dashboard/history', icon: '📋' }
      ]
    },
    {
      title: 'Mobile Lines',
      items: [
        { label: 'Buy an eSIM', href: '/dashboard/rentals/esim', icon: '📱' },
        { label: 'My eSIM Lines', href: '/dashboard/rentals/esim/mine', icon: '📶' },
        { label: 'Recharge a Line', href: '/dashboard/rentals/esim/recharge', icon: '🔄' }
      ]
    },
    {
      title: 'Communication',
      items: [
        { label: 'Phone & Dialer', href: '/dashboard/phone', icon: '📞' },
        { label: 'SMS Messages', href: '/dashboard/sms', icon: '💬' }
      ]
    },
    {
      title: 'Account',
      items: [
        { label: 'Wallet & Top Up', href: '/dashboard/wallet', icon: '💳' },
        { label: 'Price List', href: '/dashboard/pricelist', icon: '🏷️' }
      ]
    }
  ];

  if (isAdmin) {
    navGroups.push({
      title: 'Administration',
      items: [
        { label: 'Admin Desk', href: '/admin/rentals', icon: '🛠️' }
      ]
    });
  }

  const handleSignOut = async () => {
    await supabase.auth.signOut();
    router.push('/login');
  };

  return (
    <div className="min-h-screen bg-[#070c18] text-slate-100 flex flex-col md:flex-row font-sans">
      {/* Mobile Top Header */}
      <div className="md:hidden bg-[#0b1329] border-b border-slate-800 p-4 flex items-center justify-between sticky top-0 z-40">
        <Link href="/dashboard" className="flex items-center gap-2">
          <div className="w-8 h-8 bg-emerald-500 rounded-lg flex items-center justify-center font-black text-black text-lg">N</div>
          <span className="font-black tracking-tight text-white text-lg">NAVA</span>
        </Link>

        <div className="flex items-center gap-3">
          <div className="bg-slate-900 border border-slate-800 px-3 py-1 rounded-lg text-xs font-mono font-bold text-emerald-400">
            ${balance}
          </div>
          <button 
            onClick={() => setMobileOpen(!mobileOpen)}
            className="p-2 text-slate-400 hover:text-white bg-slate-900 border border-slate-800 rounded-lg"
          >
            {mobileOpen ? '✕' : '☰'}
          </button>
        </div>
      </div>

      {/* Sidebar Navigation */}
      <aside className={`
        fixed md:static inset-y-0 left-0 z-50 w-64 bg-[#080e1e] border-r border-slate-800/80 flex flex-col justify-between transition-transform duration-200 ease-in-out
        ${mobileOpen ? 'translate-x-0' : '-translate-x-full md:translate-x-0'}
      `}>
        <div className="p-6 space-y-6 overflow-y-auto max-h-[calc(100vh-80px)] md:max-h-none">
          {/* Logo */}
          <Link href="/dashboard" className="flex items-center gap-2.5">
            <div className="w-9 h-9 bg-emerald-500 rounded-xl flex items-center justify-center font-black text-black text-xl shadow-lg shadow-emerald-950/50">N</div>
            <span className="font-black tracking-tight text-white text-xl">NAVA</span>
          </Link>

          {/* Grouped Links */}
          <div className="space-y-6 pt-2">
            {navGroups.map((group, idx) => (
              <div key={idx} className="space-y-2">
                <h3 className="text-[10px] font-bold text-slate-500 uppercase tracking-widest px-3">
                  {group.title}
                </h3>
                <div className="space-y-1">
                  {group.items.map((item) => {
                    const isActive = pathname === item.href || (item.href !== '/dashboard' && pathname?.startsWith(item.href));
                    return (
                      <Link
                        key={item.href}
                        href={item.href}
                        onClick={() => setMobileOpen(false)}
                        className={`
                          flex items-center gap-3 px-3 py-2 rounded-xl text-xs font-semibold transition-all
                          ${isActive 
                            ? 'bg-emerald-500/10 text-emerald-400 border border-emerald-500/20 font-bold' 
                            : 'text-slate-400 hover:text-white hover:bg-slate-900/60'
                          }
                        `}
                      >
                        <span className="text-sm">{item.icon}</span>
                        <span>{item.label}</span>
                      </Link>
                    );
                  })}
                </div>
              </div>
            ))}
          </div>
        </div>

        {/* User Footer / Balance */}
        <div className="p-4 border-t border-slate-800/80 bg-[#060b18] space-y-3">
          <div className="bg-[#0b1329] border border-slate-800 rounded-xl p-3 flex items-center justify-between">
            <div className="truncate pr-2">
              <div className="text-[10px] text-slate-500 font-bold uppercase tracking-wider">Wallet Balance</div>
              <div className="text-sm font-black text-emerald-400 font-mono">${balance}</div>
            </div>
            <Link 
              href="/dashboard/wallet"
              className="bg-emerald-600 hover:bg-emerald-500 text-white text-[10px] font-bold px-2.5 py-1.5 rounded-lg transition-all"
            >
              Top Up
            </Link>
          </div>

          <div className="flex items-center justify-between text-xs px-1 text-slate-500">
            <span className="truncate max-w-[130px]" title={userEmail}>{userEmail || 'Account'}</span>
            <button 
              onClick={handleSignOut}
              className="text-slate-400 hover:text-red-400 text-[11px] font-semibold transition-colors"
            >
              Sign Out
            </button>
          </div>
        </div>
      </aside>

      {/* Main Content Viewport */}
      <main className="flex-1 bg-[#070c18] overflow-y-auto">
        {children}
      </main>
    </div>
  );
}
'@
