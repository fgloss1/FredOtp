import { NextRequest, NextResponse } from 'next/server';
import { createClient } from '@supabase/supabase-js';
import { getPlanById } from '@/lib/esim/plans';
import { calculatePlanQuote } from '@/lib/esim/pricing';

export async function POST(req: NextRequest) {
  try {
    const authHeader = req.headers.get('authorization');
    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      return NextResponse.json({ error: 'Unauthorized: Session missing' }, { status: 401 });
    }

    const token = authHeader.replace('Bearer ', '').trim();
    const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL || '';
    const supabaseAnonKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY || '';
    const serviceRoleKey = process.env.SUPABASE_SERVICE_ROLE_KEY || process.env.SUPABASE_SECRET_KEY;

    if (!serviceRoleKey) {
      console.error('Missing SUPABASE_SERVICE_ROLE_KEY or SUPABASE_SECRET_KEY environment variable');
      return NextResponse.json({ error: 'Server configuration error: Missing service role key' }, { status: 500 });
    }

    const supabaseAnon = createClient(supabaseUrl, supabaseAnonKey);
    const { data: { user }, error: userError } = await supabaseAnon.auth.getUser(token);

    if (userError || !user) {
      return NextResponse.json({ error: 'Unauthorized: Invalid or expired session' }, { status: 401 });
    }

    const body = await req.json();
    const { planId, zipCode, idempotencyKey: clientKey } = body;

    if (!planId) {
      return NextResponse.json({ error: 'Plan ID required' }, { status: 400 });
    }

    const db = createClient(supabaseUrl, serviceRoleKey);

    // Fetch authoritative plan from DB or local catalog fallback
    let plan: any = null;
    const { data: dbPlan } = await db
      .from('nava_esim_plans')
      .select('*')
      .or(`id.eq.${planId},code.eq.${planId}`)
      .single();

    if (dbPlan) {
      plan = dbPlan;
    } else {
      plan = getPlanById(planId);
    }

    if (!plan || plan.is_purchasable === false) {
      return NextResponse.json({ error: 'The requested plan is unavailable or coming soon.' }, { status: 400 });
    }

    // Authoritative pricing calculation
    const quote = calculatePlanQuote(plan, false);
    const preferredZip = (zipCode || '33917').replace(/\D/g, '').slice(0, 5) || '33917';

    // Reliable idempotency key
    const idempotencyKey = clientKey || `act_${user.id}_${plan.id}_${preferredZip}_${Math.floor(Date.now() / 300000)}`;

    // Check for existing order with this idempotency key
    const { data: existingOrder } = await db
      .from('nava_esim_orders')
      .select('id, status')
      .eq('user_id', user.id)
      .eq('idempotency_key', idempotencyKey)
      .maybeSingle();

    if (existingOrder) {
      return NextResponse.json({ orderId: existingOrder.id, success: true, idempotent: true });
    }

    // Insert order with all required fields
    const orderPayload: Record<string, any> = {
      user_id: user.id,
      plan_id: plan.id || plan.code || planId,
      order_type: 'activation',
      zip_code: preferredZip,
      preferred_zip: preferredZip,
      plan_price_cents: quote.plan_price_cents,
      regulatory_fee_cents: quote.regulatory_fee_cents,
      tax_cents: quote.tax_cents,
      total_cents: quote.total_cents,
      currency: 'USD',
      idempotency_key: idempotencyKey,
      status: 'pending_payment',
      created_at: new Date().toISOString(),
      updated_at: new Date().toISOString()
    };

    const { data: order, error: insertError } = await db
      .from('nava_esim_orders')
      .insert(orderPayload)
      .select('id')
      .single();

    if (insertError || !order) {
      console.error('eSIM Order Insert Error:', insertError);
      return NextResponse.json({ error: insertError?.message || 'Failed to create order' }, { status: 500 });
    }

    return NextResponse.json({ orderId: order.id, success: true });
  } catch (err: any) {
    console.error('Order Exception:', err);
    return NextResponse.json({ error: err.message || 'Internal server error' }, { status: 500 });
  }
}
