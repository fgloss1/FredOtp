$rechargeTsx = @'
'use client';

import React, { useState } from 'react';
import Link from 'next/link';
import { createClient } from '@supabase/supabase-js';

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL || '',
  process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY || ''
);

export default function EsimRechargePage() {
  const [phoneNumber, setPhoneNumber] = useState('');
  const [searching, setSearching] = useState(false);
  const [line, setLine] = useState<any>(null);
  const [errorMsg, setErrorMsg] = useState('');
  const [recharging, setRecharging] = useState(false);

  const handleLookup = async (e: React.FormEvent) => {
    e.preventDefault();
    const cleanNum = phoneNumber.replace(/\D/g, '');
    if (cleanNum.length < 10) {
      setErrorMsg('Please enter a valid 10-digit US phone number.');
      return;
    }

    setSearching(true);
    setErrorMsg('');
    setLine(null);

    try {
      const { data: { session } } = await supabase.auth.getSession();
      if (!session) {
        setErrorMsg('Please log in to recharge a line.');
        setSearching(false);
        return;
      }

      // Query user's matching active order
      const { data: orders, error } = await supabase
        .from('nava_esim_orders')
        .select('*')
        .eq('user_id', session.user.id)
        .or(`assigned_phone_number.ilike.%${cleanNum}%,phone_number.ilike.%${cleanNum}%`)
        .order('created_at', { ascending: false });

      if (error || !orders || orders.length === 0) {
        setErrorMsg('No matching active NAVA line found for this phone number under your account.');
      } else {
        setLine(orders[0]);
      }
    } catch {
      setErrorMsg('Failed to lookup phone number.');
    } finally {
      setSearching(false);
    }
  };

  const handleConfirmRecharge = async () => {
    if (!line) return;
    setRecharging(true);
    setErrorMsg('');

    try {
      const { data: { session } } = await supabase.auth.getSession();
      if (!session) throw new Error('Session expired. Please log in again.');

      const res = await fetch(`/api/esim/orders/${line.id}/renew`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${session.access_token}`
        }
      });

      const data = await res.json();
      if (!res.ok) throw new Error(data.error || 'Recharge request failed.');

      alert('Recharge request submitted successfully! Your renewal will be processed.');
      setLine(null);
      setPhoneNumber('');
    } catch (err: any) {
      setErrorMsg(err.message || 'Recharge processing failed.');
    } finally {
      setRecharging(false);
    }
  };

  const formatDate = (dateStr?: string) => {
    if (!dateStr) return 'N/A';
    return new Date(dateStr).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
  };

  const getNewExpiryDate = (currentExpiry?: string) => {
    const base = currentExpiry && new Date(currentExpiry).getTime() > Date.now() 
      ? new Date(currentExpiry) 
      : new Date();
    const extended = new Date(base.getTime() + 30 * 24 * 60 * 60 * 1000);
    return formatDate(extended.toISOString());
  };

  return (
    <div className="p-6 max-w-4xl mx-auto space-y-8">
      {/* Breadcrumbs */}
      <div className="flex items-center gap-2 text-xs font-semibold text-slate-400 uppercase tracking-wider">
        <Link href="/dashboard" className="hover:text-white transition-colors">Dashboard</Link>
        <span>/</span>
        <Link href="/dashboard/rentals/esim/mine" className="hover:text-white transition-colors">My eSIM Lines</Link>
        <span>/</span>
        <span className="text-emerald-400">Recharge Line</span>
      </div>

      {/* Header */}
      <div>
        <h1 className="text-2xl font-black text-white tracking-tight">Recharge a Dedicated US Line</h1>
        <p className="text-xs text-slate-400 mt-1">Add 30 days of service to any active NAVA mobile line registered to your account.</p>
      </div>

      {/* Lookup Card */}
      <div className="bg-[#0b1329] border border-slate-800/80 rounded-2xl p-6 sm:p-8 space-y-6 shadow-2xl">
        <form onSubmit={handleLookup} className="space-y-4">
          <label className="text-xs font-bold text-slate-300 uppercase tracking-wider block">
            Enter Your NAVA US Phone Number
          </label>
          <div className="flex flex-col sm:flex-row gap-3">
            <input 
              type="text"
              placeholder="+1 (339) 555-0199 or 3395550199"
              value={phoneNumber}
              onChange={(e) => setPhoneNumber(e.target.value)}
              className="flex-1 bg-slate-950 border border-slate-800 focus:border-emerald-500/50 rounded-xl px-4 py-3 text-white font-mono text-base outline-none transition-all"
            />
            <button
              type="submit"
              disabled={searching || !phoneNumber}
              className="bg-emerald-600 hover:bg-emerald-500 disabled:bg-slate-800 disabled:text-slate-500 text-white font-bold px-6 py-3 rounded-xl text-sm transition-all shadow-lg shadow-emerald-950/40"
            >
              {searching ? 'Looking up...' : 'Verify Line'}
            </button>
          </div>
          <p className="text-[10px] text-slate-500">Only mobile numbers purchased through your NAVA account can be looked up and recharged.</p>
        </form>

        {/* Error Alert */}
        {errorMsg && (
          <div className="p-4 bg-red-950/40 border border-red-800/50 rounded-xl text-xs text-red-400 font-medium">
            {errorMsg}
          </div>
        )}

        {/* Line Verified Card */}
        {line && (
          <div className="bg-[#0f1b38] border border-emerald-500/30 rounded-2xl p-6 space-y-6 animate-fadeIn">
            <div className="flex items-center justify-between pb-4 border-b border-slate-800">
              <div>
                <span className="text-[10px] font-bold text-emerald-400 uppercase tracking-wider block mb-0.5">Line Verified</span>
                <h3 className="text-2xl font-black text-white font-mono tracking-tight">
                  {line.assigned_phone_number || line.phone_number}
                </h3>
              </div>
              <span className="px-3 py-1 bg-emerald-500/10 border border-emerald-500/30 text-emerald-400 text-xs font-bold uppercase rounded-lg">
                Active Line
              </span>
            </div>

            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 text-xs">
              <div className="bg-slate-950/60 border border-slate-800 p-3.5 rounded-xl">
                <span className="text-[10px] text-slate-500 font-bold uppercase block mb-1">Carrier Plan</span>
                <span className="text-white font-bold uppercase">{line.plan_id?.replace('_', ' ')}</span>
              </div>
